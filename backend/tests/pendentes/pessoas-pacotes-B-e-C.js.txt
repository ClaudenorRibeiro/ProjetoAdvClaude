test('ACHADO anotações: pessoa que não existe = 404 (hoje grava anotação órfã); texto que não é texto ou grande demais = 400; rota da jurídica grava como jurídica', async () => {
  const falhas = [];
  const antes = await total('SELECT COUNT(*) AS n FROM historico_atendimento');
  const r1 = await api().post('/api/pessoas/fisicas/999999/historico').send({ descricao: 'para ninguém' });
  if (r1.status !== 404) falhas.push(`pessoa inexistente → ${r1.status}`);
  const r2 = await api().post('/api/pessoas/fisicas/abc/historico').send({ descricao: 'para ninguém' });
  if (r2.status !== 404) falhas.push(`id abc → ${r2.status}`);
  if (await total('SELECT COUNT(*) AS n FROM historico_atendimento') !== antes) falhas.push('gravou anotação órfã');
  const pf = await criarPF();
  for (const descricao of ruins) { const r = await api().post(`/api/pessoas/fisicas/${pf}/historico`).send({ descricao }); if (r.status !== 400) falhas.push(`descricao ${JSON.stringify(descricao)} → ${r.status}`); }
  const enorme = await api().post(`/api/pessoas/fisicas/${pf}/historico`).send({ descricao: grande(70000) }); if (enorme.status !== 400) falhas.push(`descricao 70000 letras → ${enorme.status}`);
  const pj = await criarPJ();
  await api().post(`/api/pessoas/juridicas/${pj}/historico`).send({ descricao: 'sem tipo no corpo' });
  const tipo = await um('SELECT tipo_pessoa FROM historico_atendimento WHERE pessoa_id = ? ORDER BY id DESC LIMIT 1', [pj]);
  if (!tipo || tipo.tipo_pessoa !== 'juridica') falhas.push(`rota /juridicas/:id/historico sem tipo_pessoa no corpo gravou como "${tipo && tipo.tipo_pessoa}"`);
  const ed = await api().put('/api/pessoas/historico/1').send({ descricao: 123 }); if (ed.status >= 500) falhas.push(`editar descricao número → ${ed.status}`);
  assert.deepEqual(falhas, []);

test('ACHADO busca: "%" e "_" são procurados como texto (não como "qualquer coisa")', async () => {
  const unico = `Curinga${Date.now() % 100000}`;
  await criarPF({ nome: `${unico} Alfa` }); await criarPF({ nome: `${unico} Beta`, cidade: '100% Campinas' });
  const n = async (q) => (await api().get(`/api/pessoas/fisicas?busca=${encodeURIComponent(q)}`)).body.dados.total;

});
