test('ACHADO busca: "%" e "_" são procurados como texto (não como "qualquer coisa")', async () => {
  const unico = `Curinga${Date.now() % 100000}`;
  await criarPF({ nome: `${unico} Alfa` }); await criarPF({ nome: `${unico} Beta`, cidade: '100% Campinas' });
  const n = async (q) => (await api().get(`/api/pessoas/fisicas?busca=${encodeURIComponent(q)}`)).body.dados.total;
  assert.equal(await n('%'), 1, 'busca "%" deve achar só quem tem % no texto');
  assert.equal(await n('_'), 0, 'busca "_" não deve achar todo mundo');
  assert.equal(await n(`${unico} A_fa`), 0);
});

test('ACHADO OBSERVAÇÃO busca de telefone: telefone digitado só com números acha o gravado com máscara "(19) 98877-6655"', async () => {
  const unico = `Fone${Date.now() % 100000}`;
  await criarPF({ nome: unico, telefones: [{ numero: '(19) 98877-6655' }] });
  const r = await api().get('/api/pessoas/fisicas?busca=19988776655');
  assert.ok(r.body.dados.registros.some(x => x.nome === unico), 'não achou pelo telefone só com dígitos');
});
